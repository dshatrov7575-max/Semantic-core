## 4. Проект решения D27

**D27.1 Схема как данные.** В онтологию добавляются виды записей `ClassDef` (имя, родитель, атрибуты с типом,
кардинальностью, обязательностью; наследование вниз по дереву), `LinkDef` (тип связи: домен, диапазон,
кардинальность, симметричность), `IdentifierDef` (тип идентификатора: ИНН, ОГРН, ОГРНИП, СНИЛС…, с правилом
нормализации и проверки и признаком «сильный/слабый»; телефон — слабый) и `SchemaChange` (append-only запись изменения: кто, когда, что). Область схемы — tenant;
записи схемы маркируются как любые записи (D6). **Корневые классы** — девять типов сущностей `ac.entities` (`PERSON`,
`ORGANIZATION`, …, `CONCEPT`); нужен ли десятый общий корень `THING` для таксономий вроде «Акулы» (иначе они идут под
`CONCEPT`) — вопрос §8 п. 8; корни фиксированы релизом (это DDL — наше, однократное), пользовательские классы
наследуют от корней. **Принадлежность
сущности пользовательскому классу — обычное утверждение** (`is_a`; объект — ссылка на `ClassDef`, новый вид литерала
`CLASS_REF`, не сущность; с доказательством и временем): тип сущности остаётся неизменяемым корнем, класс можно
уточнять append-only, «класс на момент t» — по системному времени записи (`ingested_at`), как и всё ниже (D16).
Версия схемы **не входит** в тело утверждения (`claim_id` не меняется): применимая версия схемы определяется по тому
же системному времени записи (`ingested_at`, как у стражей ядра — иначе утверждением «задним числом» можно выбрать
старую схему) относительно журнала `SchemaChange`. Миграции: неломающие (добавить класс/атрибут,
переименовать показ) — просто запись; ломающие (сменить тип, удалить/перенести атрибут) — новая версия класса,
старые утверждения не трогаются и читаются по своей версии; «cast/move» по умолчанию — *при чтении*: проекция отображает старый
атрибут в новый по журналу схем, новых утверждений не возникает; материализация (если нужна) — отдельный конвейер
миграции с собственным артефактом и receipt (D7, D19), его утверждения — PIPELINE, ссылаются на те же *источники*
(байты и диапазоны; узлы артефактов не делятся — `claim_evidence_one_node`) и на старое утверждение как на
основание; авторство HUMAN-утверждений не переносится — старое остаётся со статусом `WITHDRAWN` через
`claim_reviews`, и существующий страж не даст сделать это под закрытой Проверкой. Иерархия — closure-таблица классов, маленькая, пересчитывается при изменении
схемы (это материализация *схемы*, не данных). Валидатор получает фазу «утверждение соответствует схеме своей
версии» и коды (по правилу «код — вектор — мутант»). «Конструктор модели» — клиент реестра (интерфейс — в
SemanticWiki; для ChatGPT и скриптов — те же записи через валидатор).

Циклы (предложение; порядок — ваше решение, §8):

  модели как CLI над записями,
  векторы/мутанты, независимая рецензия.

## Проверил и дефекта не нашёл

- Одиннадцать векторов цикла 9 (P801, P802, N801–N803, N810, N811, N820, N830–N832) дают ровно заявленные коды; контрольные мутанты (аналог M830 и «не проверяется range») убиваются.
- Цикл наследования в валидаторе обнаруживается: самоссылка и A→B→A дают SCHEMA_INVALID; цепочка из 400 классов валидна и проверяется за 0,3 с.
- В базе цикл построить нельзя по построению: родитель обязан существовать в момент вставки, записи неизменяемы; самоссылка и взаимная ссылка A↔B одним INSERT дают REF_UNRESOLVED (`extra3_s9.out`). - Closure после последовательных вставок (плоские наборы, цепочки, наследники) совпадает с рекурсивным расчётом по `class_defs` (§9, «closure == рекурсивный расчёт: true»).

- TRUNCATE под `ac_loader` — `permission denied`; отдельной защиты от TRUNCATE нет и у прежних таблиц, отличия от принятой в проекте практики нет.

-- tenant isolation: parent must be in the same tenant
CREATE FUNCTION ac.class_defs_guard() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  NEW.created_at := CASE WHEN NEW.created_at IS NOT NULL THEN NEW.created_at ELSE clock_timestamp() END;
  IF NEW.parent_class_id IS NOT NULL THEN
    -- parent must exist within the same tenant
    IF NOT EXISTS (SELECT 1 FROM ac.class_defs p
                   WHERE p.class_id = NEW.parent_class_id AND p.tenant_id = NEW.tenant_id) THEN
      PERFORM ac.fail('REF_UNRESOLVED',
        'parent_class_id ' || NEW.parent_class_id || ' не найден в tenant ' || NEW.tenant_id);
    END IF;
    -- root_type of child must match root_type of parent
    PERFORM 1 FROM ac.class_defs p
    WHERE p.class_id = NEW.parent_class_id AND p.root_type <> NEW.root_type;
    IF FOUND THEN
      PERFORM ac.fail('SCHEMA_INVALID',
        'root_type ' || NEW.root_type || ' наследника не совпадает с root_type родителя');
    END IF;
    -- cycle detection: NEW.class_id must not appear in any ancestor chain
    IF EXISTS (
      WITH RECURSIVE chain(cid) AS (
        SELECT NEW.parent_class_id
        UNION ALL
        SELECT p.parent_class_id FROM ac.class_defs p JOIN chain c ON p.class_id = c.cid
        WHERE p.parent_class_id IS NOT NULL
      )
      SELECT 1 FROM chain WHERE cid = NEW.class_id
    ) THEN
      PERFORM ac.fail('SCHEMA_INVALID', 'циклическое наследование: ' || NEW.class_id);
    END IF;
  END IF;
  RETURN NEW;
END $$;

-- ---------------------------------------------------------------- class_closure (транзитивное замыкание parent_class_id)
-- Строки: (ancestor_id, descendant_id, depth).  Каждый класс — собственный предок (depth=0).
-- Пересчитывается полностью по tenant при каждом INSERT в ac.class_defs, т.к. записи неизменяемы —

CREATE TABLE ac.class_closure (
  ancestor_id   text    NOT NULL,
  descendant_id text    NOT NULL,
  depth         integer NOT NULL CHECK (depth >= 0),
  tenant_id     text    NOT NULL,
  PRIMARY KEY (ancestor_id, descendant_id)
);
CREATE INDEX class_closure_desc ON ac.class_closure (descendant_id);

CREATE FUNCTION ac.rebuild_class_closure(p_tenant text) RETURNS void LANGUAGE sql AS $$
  DELETE FROM ac.class_closure WHERE tenant_id = p_tenant;
  INSERT INTO ac.class_closure (ancestor_id, descendant_id, depth, tenant_id)
  WITH RECURSIVE closure(ancestor_id, descendant_id, depth) AS (
    SELECT class_id, class_id, 0 FROM ac.class_defs WHERE tenant_id = p_tenant
    UNION ALL
    SELECT c.ancestor_id, d.class_id, c.depth + 1
    FROM closure c
    JOIN ac.class_defs d ON d.parent_class_id = c.descendant_id AND d.tenant_id = p_tenant
  )
  SELECT DISTINCT ancestor_id, descendant_id, depth, p_tenant FROM closure;
$$;

BEGIN
  PERFORM ac.rebuild_class_closure(NEW.tenant_id);
  RETURN NULL;
END $$;

  FOR EACH ROW EXECUTE FUNCTION ac.class_defs_closure_refresh();

END $$;

-- When a Claim with predicate 'schema.is_a' is inserted, enforce:
--   1. object must be a JSONB literal of type CLASS_REF: {"type":"CLASS_REF","class_id":"sdf_...","tenant_id":"tnt_..."}
--   2. class_id must exist in ac.class_defs within the same tenant
--   3. entity_type of subject entity must match ClassDef.root_type

CREATE FUNCTION ac.claims_isa_guard() RETURNS trigger LANGUAGE plpgsql AS $$
DECLARE
  obj      jsonb;
  cid      text;
  cls      ac.class_defs;
  ent_type text;
  prj_tnt  text;
BEGIN
  IF NEW.predicate <> 'schema.is_a' THEN
    RETURN NEW;
  END IF;
  obj := NEW.body -> 'object' -> 'literal';
  IF obj IS NULL OR obj ->> 'type' <> 'CLASS_REF' THEN
    PERFORM ac.fail('PREDICATE_RANGE_VIOLATION',
      'schema.is_a: объект должен быть литералом CLASS_REF, получено: ' || (NEW.body->'object')::text);
  END IF;
  cid := obj ->> 'class_id';
  -- resolve tenant from project
  SELECT p.tenant_id INTO prj_tnt FROM ac.projects p WHERE p.project_id = NEW.project_id;
  SELECT * INTO cls FROM ac.class_defs WHERE class_id = cid;
  IF cls.class_id IS NULL THEN
    PERFORM ac.fail('REF_UNRESOLVED',
      'schema.is_a: class_id ' || coalesce(cid, '<null>') || ' не найден');
  END IF;
  IF prj_tnt IS NOT NULL AND cls.tenant_id <> prj_tnt THEN
    PERFORM ac.fail('CROSS_SCOPE_REFERENCE',
      'schema.is_a: ClassDef из tenant ' || cls.tenant_id || ', проект в tenant ' || prj_tnt);
  END IF;
  SELECT e.entity_type INTO ent_type FROM ac.entities e WHERE e.entity_id = NEW.subject;
  IF ent_type IS NOT NULL AND ent_type <> cls.root_type THEN
    PERFORM ac.fail('PREDICATE_DOMAIN_VIOLATION',
      'schema.is_a: entity_type ' || ent_type || ' != ClassDef.root_type ' || cls.root_type);
  END IF;
  RETURN NEW;
END $$;
